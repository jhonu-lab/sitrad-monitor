"""

No tengo ese archivo, así que no lo toqué: copia este método tal cual. Código nuevo; si nadie lo llama no cambia nada.
Requiere en el módulo: `import logging`, `import requests` y `logger = logging.getLogger(__name__)`.
"""

    def send_photo(self, path: str, caption: str = "", chat_id=None) -> bool:
        """Envía una imagen (PNG/JPG) al chat. True si Telegram la aceptó; nunca lanza por errores de red."""
        target = chat_id or getattr(self, "chat_id", None)
        if not target:
            logger.warning("send_photo: no hay chat_id.")
            return False
        try:
            with open(path, "rb") as fh:
                r = self._session.post(
                    f"{self.base_url}/sendPhoto",
                    data={"chat_id": target, "caption": (caption or "")[:1024]},
                    files={"photo": fh},
                    timeout=self.send_timeout,
                )
            if not r.ok:
                logger.warning("sendPhoto falló: HTTP %s", r.status_code)
            return bool(r.ok)
        except (requests.RequestException, OSError) as exc:
            logger.warning("sendPhoto falló: %s", type(exc).__name__)    # sin la URL: contiene el token
            return False
